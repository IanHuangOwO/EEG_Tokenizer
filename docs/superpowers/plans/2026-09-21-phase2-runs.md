> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run the Phase 2 cross-subject experiments on the frozen `mesae_v10_small` backbone and record them in ADR 0014: (1) PhysionetMI, C1 vs raw, one model trained on 30 unseen subjects and evaluated on a seen group and a representative unseen group; (2) BETA_4s (SSVEP), C1 vs C1+phase advance (C3), same design, next to the zero-training PSDA reference; (3) Inria (ERP), C1 (induced) vs C4 (evoked) vs raw ERP, grouped 4-fold by subject.

**Architecture:** No new model code (50 epochs per run). A generator script writes seven finetune configs into `config/phase2/`; each is run with `train_finetune.py` (`split_mode: "subject_groups"`, added in commit 71f952d); `probes/group_summary.py` (a79987f, fixed in f1e60e6) turns each run's `group_eval.json` into per-subject paired comparisons. A small `probes/epoch_curve.py` reads the training logs to check convergence, because Phase 1 showed 30 epochs was still rising slightly.

**Tech Stack:** `train_finetune.py`, `probes/group_summary.py`, `probes/phase_probe_beta.py` (PSDA reference), scipy.

**Spec:** `docs/adr/0014-finetune-head-test-plan.md` (Protocol: subject is the unit of analysis, tail-mean = mean of the last 10 epochs, point difference is the bar and p is reported never gated; step 11 incl. the "LOSO, MI core (phase 1)" entry; C3 = step 10, C4 = step 12). ## Cost model (calibrated: about 10.2 ms per train+val sample per epoch for C1, 0.71x for raw)

| Run | Train + val trials | Seconds per epoch | Hours at 50 epochs |
|---|---|---|---|
| PhysionetMI C1 (30 train + 13 eval subjects, about 363 trials each) | about 15,600 | about 159 | 2.2 |
| PhysionetMI raw | same | about 113 | 1.6 |
| BETA_4s C1 or C1+advance (30 train + 13 eval, 160 trials each) | about 6,900 | about 70 | 1.0 each |
| Inria, per fold (9 train + 3 heldout subjects of a 12-subject subset, 340 trials each) | about 4,080 | about 42 | 0.6 per fold, 2.3 per head over 4 folds (raw about 1.7) |

Total at 50 epochs: about 3.8 + 2.0 + 6.3 = 12 GPU-hours, all sequential. The tail evaluation (last 10 epochs, per subject) adds a few percent.

## Shared conventions

- Run directories: `output/mesae_p2_<dataset>_<tag>/` with `artifacts/group_eval.json` and `artifacts/train_*.log`. `model_name` in each config is that directory name.
- Comparison command (first file = reference): `python probes/group_summary.py output/<ref>/artifacts/group_eval.json output/<other>/artifacts/group_eval.json`.
- Fold/subject naming inside `group_eval.json`: PhysionetMI and BETA_4s have one run named `main` with groups `seen` and `unseen`; Inria has runs `fold0` to `fold3`, each with group `heldout` (pooled per subject by `group_summary.py`).

- [ ] **Step 1: Write `probes/make_phase2_configs.py`**

```python
"""Write the Phase 2 finetune configs (docs/superpowers/plans/2026-09-21-phase2-runs.md).
Usage: python probes/make_phase2_configs.py [--out config/phase2] [--epochs 50]
(--epochs is the default; EPOCHS below overrides it per dataset)
Each config = config/config.json with the finetune dataset, head, and split replaced."""
import argparse, copy, json, os

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

C1 = dict(input='stamp_induced', pool_channel='spatial:8', pool_time='learned:2', dropout=0.5,
          freeze_backbone=True)
HEADS = {
    'c1': C1,
    'c3': dict(C1, include_advance=True),
    'c4': dict(C1, evoked_rank=2),
    'raw': dict(input='raw', pool_channel='spatial:8', pool_time='trial', dropout=0.5,
                freeze_backbone=True),
}
EPOCHS = {'inria': 30}  # shorter to save GPU time; convergence is checked afterwards
# (dataset dir, short name, subject_groups json or None, task, split, head tags)
PLAN = [
    ('PhysionetMI', 'eegmmidb', 'eegmmidb', 'mi', 'groups', ['c1', 'raw']),
    ('BETA_4s', 'beta4s', 'beta4s', 'ssvep', 'groups', ['c1', 'c3']),
    ('Inria_Train', 'inria', None, 'erp', 'kfold4', ['c1', 'c4', 'raw']),
]


def build(base, ds, short, groups_key, task, split, tag, epochs):
    cfg = copy.deepcopy(base)
    cfg['dataset_params']['finetune'] = {
        ds: {'dataset_path': f'datas/{ds}', 'subject_to_use': ['all'], 'channels_to_use': ['all']}}
    cfg['model_params']['MeSAE']['finetune'] = dict(HEADS[tag], task=task)
    tp = cfg['training_params']['finetune']
    tp.pop('cv_folds', None)
    tp.update(model_name=f'mesae_p2_{short}_{tag}', epochs=EPOCHS.get(short, epochs),
              split_mode='subject_groups')
    if split == 'groups':
        g = json.load(open(os.path.join(REPO, 'config', 'subject_groups', f'{groups_key}.json')))
        tp['subject_group_runs'] = [{'name': 'main', 'train': g['train'], 'eval': g['eval']}]
    else:
        tp['subject_kfold'] = 4  # subject_kfold_seed defaults to 42: same folds for every head
    return cfg


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--out', default=os.path.join(REPO, 'config', 'phase2'))
    ap.add_argument('--epochs', type=int, default=50)
    a = ap.parse_args()
    base = json.load(open(os.path.join(REPO, 'config', 'config.json')))
    os.makedirs(a.out, exist_ok=True)
    for ds, short, groups_key, task, split, tags in PLAN:
        for tag in tags:
            cfg = build(base, ds, short, groups_key, task, split, tag, a.epochs)
            name = f'{short}_{tag}.json'
            with open(os.path.join(a.out, name), 'w', newline='\n') as f:
                json.dump(cfg, f, indent=1)
                f.write('\n')
            print('wrote', os.path.join(a.out, name))


if __name__ == '__main__':
    main()
```

```python
"""Convergence check from training logs. Usage: python probes/epoch_curve.py <run_dir> [block=5]
Mean [Val]/[Train] bal_acc per <block>-epoch block over the fold tags found in
<run_dir>/artifacts/train_*.log; also prints tail (last 10) vs the previous 10 epochs."""
import glob, os, re, sys
import numpy as np

run = sys.argv[1]
block = int(sys.argv[2]) if len(sys.argv) > 2 else 5
val, trn = {}, {}
for path in sorted(glob.glob(os.path.join(run, 'artifacts', 'train_*.log'))):
    tag = ep = None
    for line in open(path):
        m = re.search(r'\[([^\]]+)\] Epoch (\d+)/(\d+) Summary', line)
        if m:
            tag, ep, total = m.group(1), int(m.group(2)), int(m.group(3))
            continue
        m = re.search(r'\[(Val|Train)\].*bal_acc: ([\d.]+)', line)
        if m and tag:
            (val if m.group(1) == 'Val' else trn).setdefault(tag, {})[ep] = float(m.group(2))
tags = [t for t in val if len(val[t]) >= total]
print(f'{run}: {len(tags)} complete fold tags, {total} epochs')
for a in range(1, total + 1, block):
    b = min(a + block - 1, total)
    v = np.mean([np.mean([val[t][e] for e in range(a, b + 1)]) for t in tags])
    r = np.mean([np.mean([trn[t][e] for e in range(a, b + 1)]) for t in tags])
    print(f' ep{a:3}-{b:3}  val {v:.3f}  train {r:.3f}')
if total >= 20:
    last = np.mean([np.mean([val[t][e] for e in range(total - 9, total + 1)]) for t in tags])
    prev = np.mean([np.mean([val[t][e] for e in range(total - 19, total - 9)]) for t in tags])
    print(f' tail(last 10) {last:.3f} vs previous 10 {prev:.3f}  gain {last - prev:+.3f}')
```
